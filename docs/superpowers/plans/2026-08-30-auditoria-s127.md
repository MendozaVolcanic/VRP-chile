```python
# -*- coding: utf-8 -*-
"""S127 — inventario de afirmaciones que el codigo hace sobre su propio estado.

POR QUE: S126 encontro dos casos donde un comentario afirmaba algo del sistema y era
falso — el PR #535 decia "no-op" y apago la mascara de nube en produccion; el docstring
de single_pixel_mode dice "Volcanes NO afectados ... Lascar" y esta activo en 110/110
records de Lascar. Los dos costaron sesiones.

No se puede verificar automaticamente si una afirmacion es cierta. Si se puede LISTARLAS
todas para revisarlas de a una, que es lo que nunca se hizo.
"""
import io, json, os, re, sys

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
ROOT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "..")

# Frases que afirman estado del sistema. Cada una costo al menos un incidente.
PATRONES = [
    (r"no[- ]?op", "afirma que algo no tiene efecto"),
    (r"NO afectad", "afirma que algo no aplica a cierto caso"),
    (r"\bremovid|\bse removio|ya no (se )?(usa|aplica|corre)", "afirma que algo se quito"),
    (r"codigo muerto|dead code|nunca (se )?(llama|corre|dispara)", "afirma que algo no corre"),
    (r"solo (afecta|aplica|toca)", "acota el alcance de algo"),
    (r"por defecto (esta |queda )?(OFF|apagad)", "afirma un default"),
    (r"identico|byte a byte|no cambia (nada|el resultado)", "afirma equivalencia"),
]
EXT = (".py", ".yaml", ".yml")
SALTAR = ("/.git/", "/_archive/", "/node_modules/")


def barrer():
    out = []
    for base, _, files in os.walk(ROOT):
        if any(s in base.replace("\\", "/") for s in SALTAR):
            continue
        for f in files:
            if not f.endswith(EXT):
                continue
            p = os.path.join(base, f)
            try:
                lineas = open(p, encoding="utf-8", errors="replace").read().splitlines()
            except OSError:
                continue
            for n, linea in enumerate(lineas, 1):
                if not re.search(r"^\s*#|\"\"\"|'''", linea) and "#" not in linea:
                    continue
                for pat, clase in PATRONES:
                    if re.search(pat, linea, re.I):
                        out.append({"archivo": os.path.relpath(p, ROOT).replace("\\", "/"),
                                    "linea": n, "clase": clase,
                                    "texto": linea.strip()[:160]})
                        break
    return out


if __name__ == "__main__":
    hall = barrer()
    print("AFIRMACIONES DE ESTADO A VERIFICAR: %d\n" % len(hall))
    por_clase = {}
    for h in hall:
        por_clase.setdefault(h["clase"], []).append(h)
    for clase, xs in sorted(por_clase.items(), key=lambda kv: -len(kv[1])):
        print("%-46s %4d" % (clase, len(xs)))
    print("\nPRIORIDAD: las que estan en pipeline/ y en profiles/ gobiernan produccion.")
    prio = [h for h in hall if h["archivo"].startswith(("pipeline/", "scripts/"))]
    print("  en pipeline/ o scripts/: %d" % len(prio))
    for h in prio[:40]:
        print("  %-52s:%-5d %s" % (h["archivo"], h["linea"], h["texto"][:80]))
    dest = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                        "01_afirmaciones_de_estado.json")
    json.dump(hall, open(dest, "w", encoding="utf-8"), indent=1, ensure_ascii=False)
    print("\npersistido en", dest)
```

```python
# -*- coding: utf-8 -*-
"""S127 — funciones y flags construidos que nadie usa, o que se usan en un solo lugar
cuando deberian estar en dos.

POR QUE: `geo_utils.get_grid_center()` se escribio en S98 para resolver D17 y nadie la
llamo nunca (28 sesiones). La corona Eq.6 quedo cableada solo en el bloque Test 1 de
VIIRS375 y el A/B midio un path que aporta 3 de 80 noches comparadas. Las dos veces el
codigo estaba bien; el problema era donde (no) se lo invocaba.
"""
import ast, io, json, os, sys
from collections import defaultdict

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
ROOT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "..")
DIRS = ("pipeline", "scripts", "frontend")


def definidas():
    out = {}
    for d in DIRS:
        base = os.path.join(ROOT, d)
        if not os.path.isdir(base):
            continue
        for b, _, fs in os.walk(base):
            for f in fs:
                if not f.endswith(".py"):
                    continue
                p = os.path.join(b, f)
                try:
                    arbol = ast.parse(open(p, encoding="utf-8", errors="replace").read())
                except SyntaxError:
                    continue
                for nodo in ast.walk(arbol):
                    if isinstance(nodo, ast.FunctionDef) and not nodo.name.startswith("_"):
                        out[nodo.name] = os.path.relpath(p, ROOT).replace("\\", "/")
    return out


def call_sites(nombres):
    cuenta = defaultdict(int)
    for b, _, fs in os.walk(ROOT):
        r = b.replace("\\", "/")
        if "/.git/" in r or "/_archive/" in r:
            continue
        for f in fs:
            if not f.endswith((".py", ".yml", ".yaml")):
                continue
            try:
                txt = open(os.path.join(b, f), encoding="utf-8", errors="replace").read()
            except OSError:
                continue
            for n in nombres:
                cuenta[n] += txt.count(n + "(")
    return cuenta


if __name__ == "__main__":
    defs = definidas()
    cuenta = call_sites(list(defs))
    # 1 aparicion = solo su propia definicion
    dormidas = sorted(n for n, c in cuenta.items() if c <= 1)
    un_solo = sorted(n for n, c in cuenta.items() if c == 2)
    print("FUNCIONES SIN NINGUN CALL SITE: %d" % len(dormidas))
    for n in dormidas:
        print("  %-44s %s" % (n, defs[n]))
    print("\nCON UN SOLO CALL SITE (revisar si deberian tener dos): %d" % len(un_solo))
    for n in un_solo[:40]:
        print("  %-44s %s" % (n, defs[n]))
    dest = os.path.join(os.path.dirname(os.path.abspath(__file__)), "02_capacidad_dormida.json")
    json.dump({"sin_call_site": {n: defs[n] for n in dormidas},
               "un_solo_call_site": {n: defs[n] for n in un_solo}},
              open(dest, "w", encoding="utf-8"), indent=1, ensure_ascii=False)
    print("\npersistido en", dest)
```

```python
# -*- coding: utf-8 -*-
"""S127 — todo archivo de datos que un script consume, debe tener quien lo refresque.

POR QUE: `build_c2ab_windows.py` consumia un CSV congelado el 2026-03-28 mientras el
snapshot vivo llegaba al 2026-08-24. Perdia 59 fechas ALERTA sin que nada avisara.
"""
import io, json, os, re, sys

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
ROOT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "..")
WF = os.path.join(ROOT, ".github", "workflows")


def consumidos():
    """Rutas .csv/.json bajo data/ que aparecen en scripts o en el pipeline."""
    pat = re.compile(r"[\"']([^\"']*data/[^\"']+\.(?:csv|json))[\"']")
    out = {}
    for d in ("scripts", "pipeline", "frontend"):
        base = os.path.join(ROOT, d)
        for b, _, fs in os.walk(base) if os.path.isdir(base) else []:
            for f in fs:
                if not f.endswith((".py", ".html", ".yml")):
                    continue
                p = os.path.join(b, f)
                txt = open(p, encoding="utf-8", errors="replace").read()
                for m in pat.findall(txt):
                    out.setdefault(m, []).append(os.path.relpath(p, ROOT).replace("\\", "/"))
    return out


def refrescados():
    txt = ""
    for f in os.listdir(WF):
        if f.endswith((".yml", ".yaml")):
            txt += open(os.path.join(WF, f), encoding="utf-8", errors="replace").read()
    return txt


if __name__ == "__main__":
    cons, wf = consumidos(), refrescados()
    huerfanos = {k: v for k, v in cons.items() if os.path.basename(k) not in wf}
    print("ARCHIVOS DE DATOS CONSUMIDOS: %d   sin workflow que los refresque: %d\n"
          % (len(cons), len(huerfanos)))
    for k, v in sorted(huerfanos.items()):
        p = os.path.join(ROOT, k)
        existe = "existe" if os.path.exists(p) else "NO EXISTE"
        print("  %-64s [%s]  <- %s" % (k, existe, ", ".join(sorted(set(v)))[:60]))
    dest = os.path.join(os.path.dirname(os.path.abspath(__file__)), "03_datos_consumidos.json")

              open(dest, "w", encoding="utf-8"), indent=1, ensure_ascii=False)
```

```python
# -*- coding: utf-8 -*-
"""S127 — guard: ningun script ACTIVO consume un CSV que nadie refresca.

POR QUE: build_c2ab_windows.py leia un CSV congelado en marzo como si fuera el vivo.
Los snapshots de experiments/ SI deben quedar pinneados (A17); lo que no puede pasar es
que un script de scripts/ o pipeline/ dependa de un archivo huerfano.

import re
from pathlib import Path

REPO = Path(__file__).resolve().parents[1]
PAT = re.compile(r"[\"']([^\"']*data/[^\"']+\.csv)[\"']")

# Pinneados a proposito: snapshots historicos que deben quedar congelados (A17).
PINNEADOS_A_PROPOSITO = {
    "data/mirova_reference/registro_vrp_ocr.csv",   # lo usan experiments/126-131
}


def test_los_scripts_activos_no_consumen_csv_huerfanos():
    wf = "".join(p.read_text(encoding="utf-8", errors="replace")
                 for p in (REPO / ".github" / "workflows").glob("*.y*ml"))
    malos = []
    for d in ("scripts", "pipeline"):
        for p in (REPO / d).rglob("*.py"):
```

```python
# -*- coding: utf-8 -*-
"""S127 — que campos escribe el pipeline y cuales leen de verdad los consumidores.

POR QUE (familia A46): el piso VRP pone en cero `record.vrp_mw` mientras las 3 vistas del
frontend leen `primary_cluster.vrp_mw` — por eso "suprime 761 records" y no se ve nada.
`distance_class` sale de `final_hotspot` mientras la magnitud sale del cluster, y de ahi
salieron los flips espurios de S113. Un campo escrito que nadie lee es ruido; un campo
leido por dos consumidores con semantica distinta es un bug esperando.
"""
import io, json, os, re, sys
from collections import Counter

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
ROOT = os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "..")


def campos_en_los_datos():
    """Todas las claves que aparecen en los records operacionales."""
    c = Counter()
    d = os.path.join(ROOT, "data", "mirova_equivalent")
    for f in sorted(os.listdir(d)):
        if not f.endswith(".json"):
            continue
        for r in json.load(open(os.path.join(d, f), encoding="utf-8"))["records"]:
            c.update(r.keys())
            for sub in ("primary_cluster",):
                if isinstance(r.get(sub), dict):
                    c.update("%s.%s" % (sub, k) for k in r[sub])
    return c


def lectores(campo):
    """Cuantos archivos de frontend/, scripts/ o pipeline/ mencionan el campo."""
    hoja = campo.split(".")[-1]
    n = []
    for d in ("frontend", "scripts", "pipeline"):
        base = os.path.join(ROOT, d)
        if not os.path.isdir(base):
            continue
        for b, _, fs in os.walk(base):
            for f in fs:
                if not f.endswith((".py", ".html")):
                    continue
                p = os.path.join(b, f)
                if hoja in open(p, encoding="utf-8", errors="replace").read():
                    n.append(os.path.relpath(p, ROOT).replace("\\", "/"))
    return n


if __name__ == "__main__":
    campos = campos_en_los_datos()
    filas = []
    for campo, veces in campos.most_common():
        ls = lectores(campo)
        filas.append({"campo": campo, "records": veces, "lectores": ls})
    huerfanos = [f for f in filas if not f["lectores"]]
    print("CAMPOS EN LOS RECORDS: %d   sin ningun lector: %d\n" % (len(filas), len(huerfanos)))
    print("SIN LECTOR (se escriben y nadie los usa):")
    for f in huerfanos:
        print("  %-40s %7d records" % (f["campo"], f["records"]))
    print("\nCON MAS DE 4 LECTORES (candidatos a semantica divergente):")
    for f in filas:
        if len(f["lectores"]) > 4:
            print("  %-40s %d lectores" % (f["campo"], len(f["lectores"])))
    dest = os.path.join(os.path.dirname(os.path.abspath(__file__)), "04_escrito_vs_leido.json")
    json.dump(filas, open(dest, "w", encoding="utf-8"), indent=1, ensure_ascii=False)
    print("\npersistido en", dest)
```

- [ ] **Step 1: Medir el offset del ancla contra la grilla real de MIROVA**

D15 estableció que el `Distancia_km` de MIROVA está cuantizado a celdas de su grilla, y
S124 identificó el GeoTIFF del archivo como la grilla real (134×134). Con los TIFs ya
disponibles se puede medir, por volcán, la distancia entre **nuestro** ancla
(`vent_lat/lon`) y el **centro de la celda** de MIROVA que lo contiene.

```python
# -*- coding: utf-8 -*-
"""S127 — cuanto se corre nuestra ancla respecto del centro de celda de MIROVA.

POR QUE: D17 quedo con la premisa probada (6 de 11 volcanes con la referencia desplazada
mas de media celda) y la consecuencia sin investigar. A82 fue REBAJADA en S125 porque la
auditoria S114 nunca miro el eje geometrico — asi que una llegada por geometria NO esta
bloqueada por el "irreducible" de S114.
"""
import csv, io, json, math, os, sys

import rasterio

sys.path.insert(0, os.path.join(os.path.dirname(os.path.abspath(__file__)), ".."))
from _s126_lib import VENTS   # noqa: E402

sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding="utf-8")
SCRATCH = os.environ.get("S126_SCRATCH", "")

print("%-22s %14s %16s %14s" % ("volcan", "celda km", "offset ancla km", "% de celda"))
res = {}
idx = [r for r in csv.DictReader(open(os.path.join(SCRATCH, "index_remoto.csv"),
                                      encoding="utf-8", errors="replace"))
       if r["sensor"] == "VIIRS375"]
for vol, vent in VENTS.items():
    sel = [r for r in idx if r["volcano"].replace("Nevadosde", "") in vol or r["volcano"] == vol]
    if not sel:
        continue
    p = os.path.join(SCRATCH, "tifs", os.path.basename(sel[0]["tif_path"]))
    if not os.path.exists(p):
        continue
    with rasterio.open(p) as ds:
        tr = ds.transform
        col, fil = ~tr * (vent[1], vent[0])
        cx, cy = tr * (math.floor(col) + 0.5, math.floor(fil) + 0.5)
        km_px = abs(tr.a) * 111.32 * math.cos(math.radians(vent[0]))
        off = math.hypot((cy - vent[0]) * 111.32,
                         (cx - vent[1]) * 111.32 * math.cos(math.radians(vent[0])))
    res[vol] = {"celda_km": round(km_px, 3), "offset_km": round(off, 3),
                "pct_celda": round(100 * off / km_px, 1)}
    print("%-22s %14.3f %16.3f %13.1f%%" % (vol, km_px, off, res[vol]["pct_celda"]))

dest = os.path.join(os.path.dirname(os.path.abspath(__file__)), "01_ancla_de_la_grilla.json")
json.dump(res, open(dest, "w", encoding="utf-8"), indent=2, ensure_ascii=False)
print("\npersistido en", dest)
```
