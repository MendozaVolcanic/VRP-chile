function isValidDetection(r) {
  // S139 (decisión S138-H, hallazgo P5 del verificador S138): con cúmulo, detección válida
  // exige que el cúmulo tenga energía, que es la misma magnitud que publica mirovaEqVrp. Antes
  // bastaba `triggered_test1`, y un cúmulo en 0,0 MW contaba en el contador del cráter, las
  // pastillas y la corroboración mientras el gráfico lo mostraba en cero (familia A46). Medido
  // S139: 2.219 records; en noches de volcán sólo 161 se quedan sin detección y MIROVA publicó
  // en 1. Guard: tests/test_isvaliddetection_coherencia_s139.py. Records legacy sin cúmulo
  // (pre-S27) conservan la regla antigua.
  if (r?.primary_cluster) return (r.primary_cluster.vrp_mw ?? 0) > 0;
  if ((r?.vrp_mw ?? 0) > 0) return true;        // detección eruption-path / suma indistinta legacy
  if (r?.triggered_test1 === true) return true;  // detección Test 1 integrated-ROI (Coppola 2015)
  return false;
}

function isSummitDetection(r) {
  // S29 fix bug 8: si el record fue descartado por el pipeline (vrp_mw=0 y
  // discarded_reason no nulo), NO contar como detección summit aunque el
  // distance_class lo diga — el cluster que disparó es el descartado.
  if ((r.vrp_mw ?? 0) === 0 && r.discarded_reason && !r.triggered_test1) return false;
  if (r.distance_class === "summit") return true;
  if (r.distance_class === "far") return false;
  return (r.vrp_vent_mw ?? 0) > 0;
}
